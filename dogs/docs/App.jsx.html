import "./styles.css";
import { useState, useEffect } from "react";

/**
 * Componente principal de la aplicación que muestra una imagen aleatoria de un perro.
 * Permite al usuario obtener una nueva imagen pulsando un botón.
 * @component
 * @returns {JSX.Element} El componente de la aplicación.
 */
export default function App() {
  // Estado para controlar si la aplicación está cargando la imagen
  const [isLoading, setIsLoading] = useState(true);
  // Estado para almacenar la URL de la imagen del perro
  const [imageUrl, setImageUrl] = useState(null);

  /**
   * Hook de efecto que se ejecuta cuando cambia el estado de isLoading.
   * Si isLoading es true, realiza una petición a la API para obtener una imagen aleatoria de perro.
   */
  useEffect(() => {
    /**
     * Función asíncrona que obtiene una imagen aleatoria de perro desde la API.
     */
    async function fetchData() {
      const response = await fetch("https://dog.ceo/api/breeds/image/random");

      if (response.ok) {
        const data = await response.json();
        console.log(data); // Mostramos la respuesta en consola para depuración
        setImageUrl(data.message); // Guardamos la URL de la imagen
        setIsLoading(false); // Cambiamos el estado de isLoading a false
      } else {
        console.error("Error fetching data");
      }
    }

    // Solo llamamos a la API si isLoading es true
    if (isLoading) fetchData();
  }, [isLoading]);

  // Si está cargando, mostramos un texto indicándolo
  if (isLoading) {
    return (
      <div className="App">
        <h1>Cargando...</h1>
      </div>
    );
  }

  // Renderizado principal: botón para obtener otra imagen y la imagen actual
  return (
    <div className="App">
      {/* Al hacer clic, volvemos a poner isLoading en true para cargar otra imagen */}
      <button onClick={() => setIsLoading(true)}>
        ¡Otro!
        <span role="img" aria-label="corazón">
          ❤️
        </span>
      </button>
      {/* Mostramos la imagen obtenida de la API */}
      <img src={imageUrl} alt="Random Dog" />
    </div>
  );
}
